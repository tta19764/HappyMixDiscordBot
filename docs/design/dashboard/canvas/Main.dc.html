range-preview.js · the only real script

for (const form of document.querySelectorAll("[data-range]")) {
  const { start, end } = form.elements;
  const out = form.querySelector("[data-range-preview]");
  const save = form.querySelector("[type=submit]");
  const day = form.dataset.day ?? "Today";
  const next = form.dataset.nextDay ?? "Tomorrow";
  const mins = t => +t.slice(0, 2) * 60 + +t.slice(3);
  const render = () => {
    if (!start.value || !end.value) return;
    const span = (mins(end.value) - mins(start.value) + 1440) % 1440;
    const overnight = end.value <= start.value;
    out.querySelector("[data-text]").textContent =
      `${day} ${start.value} → ${overnight ? next : day} ${end.value}`;
    out.querySelector("[data-dur]").textContent =
      `${Math.floor(span / 60)}h ${span % 60}m`;
    out.querySelector("[data-overnight]").hidden = !overnight || !span;
    out.classList.toggle("hm-preview--invalid", !span);
    save.disabled = !span;
  };
  form.addEventListener("input", render);
  render();
}
